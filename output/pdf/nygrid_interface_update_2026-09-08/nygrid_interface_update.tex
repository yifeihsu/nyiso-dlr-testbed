% 7: limits and next work
\begin{frame}{What the results support and what comes next}
\lead{A useful 2019 DC baseline and low-error compact examples}
\begin{itemize}\tightitems
\item \textbf{Main gaps:} West Central uncertainty and UPNY transfer sharing. Test bounded, shared allocation and network parameters.
\item \textbf{AC qualification:} resolve Marcy line parameters, ratings, and controls. A separate three-hour AC checkpoint passed on compact51 \emph{without Marcy}.
\item \textbf{Independent evaluation:} freeze parameters before testing new periods. Use consistent 2025 data to validate the 71-bus expansion for DLR inference.
\end{itemize}
\par\bigskip
\minor{The selected 2019 snapshots do not establish accuracy of the 2025 model or AC/DLR feasibility of the Marcy variant.}
\source{[3]. Next steps are proposed work. No new parameter fit or untouched predictive evaluation was completed in this experiment.}
\end{frame}
